def dedupe_latest(events):
    latest = {}
    first_order = []
    seen = set()

    for event in events:
        event_id = event["id"]
        revision = event["revision"]

        if event_id not in seen:
            seen.add(event_id)
            first_order.append(event_id)

        current = latest.get(event_id)
        if current is None or revision >= current["revision"]:
            latest[event_id] = event

    return [latest[event_id] for event_id in first_order]
